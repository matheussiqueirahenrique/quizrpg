  let currentQuestion = 0;

  let scores = {};

  let playerName = "";

  window.onload = function() {

  function startQuiz() {

    const input = document.getElementById('player-name');

    if (!input.value.trim()) {

      alert("Por favor, digite seu nome antes de começar!");

      return;

    }

    playerName = input.value.trim();

    document.getElementById('screen-start').classList.add('hidden');

    document.getElementById('screen-quiz').classList.remove('hidden');

    showQuestion();

  }

  function showQuestion() {

    const q = questions[currentQuestion];

    document.getElementById('question-number').innerText = `Pergunta ${currentQuestion + 1} de ${questions.length}`;

    document.getElementById('question-text').innerText = q.q;

   

    const optsContainer = document.getElementById('options-container');

    optsContainer.innerHTML = '';

    q.opts.forEach(opt => {

      const btn = document.createElement('button');

      btn.className = 'option-btn';

      btn.innerText = opt.text;

      btn.onclick = () => selectOption(opt.class);

      optsContainer.appendChild(btn);

    });

  }

  function selectOption(cls) {

    scores[cls] = (scores[cls] || 0) + 1;

    currentQuestion++;

    if (currentQuestion < questions.length) {

      showQuestion();

    } else {

      finishQuiz();

    }

  }

  function finishQuiz() {

    let winningClass = "";

    let maxScore = -1;

    for (let cls in scores) {

      if (scores[cls] > maxScore) {

        maxScore = scores[cls];

        winningClass = cls;

      }

    }

    winningClass = winningClass || "Aventureiro";

    // Salva o registro completo (Nome + Classe + Data/Hora)

    let results = JSON.parse(localStorage.getItem('quiz_results') || '[]');

   

    // Atualiza resultado se o jogador refizer o teste ou insere um novo

    const existingIndex = results.findIndex(r => r.name.toLowerCase() === playerName.toLowerCase());

    const newEntry = {

      name: playerName,

      class: winningClass,

      date: new Date().toLocaleString('pt-BR')

    };

    if (existingIndex >= 0) {

      results[existingIndex] = newEntry;

    } else {

      results.push(newEntry);

    }

    localStorage.setItem('quiz_results', JSON.stringify(results));

    document.getElementById('screen-quiz').classList.add('hidden');

    document.getElementById('screen-result').classList.remove('hidden');

    document.getElementById('final-class').innerText = winningClass;

  }

  function updatePlayersList() {

    let results = JSON.parse(localStorage.getItem('quiz_results') || '[]');

    const container = document.getElementById('public-players-list');

    if (results.length > 0) {

      container.innerHTML = results.map(r => `<span class="player-tag">👤 ${r.name}</span>`).join('');

    } else {

      container.innerHTML = '<em>Nenhum jogador registrado ainda.</em>';

    }

  }

    if (password === MASTER_PASSWORD) {

      document.getElementById('screen-start').classList.add('hidden');

      document.getElementById('screen-master').classList.remove('hidden');

      loadMasterTable();

    } else if (password !== null) {

      alert("Senha incorreta!");

    }

  }

  function loadMasterTable() {

    let results = JSON.parse(localStorage.getItem('quiz_results') || '[]');

    const tbody = document.getElementById('master-table-body');

    if (results.length === 0) {

      tbody.innerHTML = '<tr><td colspan="3" style="text-align:center;">Nenhum registro encontrado.</td></tr>';

      return;

    }

    tbody.innerHTML = results.map(r => `

      <tr>

        <td><strong>${r.name}</strong></td>

        <td><span style="color: var(--accent); font-weight: bold;">${r.class}</span></td>

        <td style="color: var(--text-dim); font-size: 12px;">${r.date}</td>

      </tr>

    `).join('');

  }

  function closeMasterPanel() {

    document.getElementById('screen-master').classList.add('hidden');

    document.getElementById('screen-start').classList.remove('hidden');

    updatePlayersList();

  }

  function exportResultsCSV() {

    let results = JSON.parse(localStorage.getItem('quiz_results') || '[]');

    if (results.length === 0) {

      alert("Não há dados para exportar!");

      return;

    }

    let csvContent = "data:text/csv;charset=utf-8,Jogador,Classe,Data_Hora\n";

    results.forEach(r => {

      csvContent += `"${r.name}","${r.class}","${r.date}"\n`;

    });

    const encodedUri = encodeURI(csvContent);

    const link = document.createElement("a");

    link.setAttribute("href", encodedUri);

    link.setAttribute("download", "resultados_quiz_rpg.csv");

    document.body.appendChild(link);

    link.click();

    document.body.removeChild(link);

  }

  function clearResults() {

    if (confirm("Tem certeza de que deseja apagar todos os resultados salvos? Esta ação não pode ser desfeita.")) {

      localStorage.removeItem('quiz_results');

      loadMasterTable();

      alert("Todos os resultados foram apagados!");

    }

  }
